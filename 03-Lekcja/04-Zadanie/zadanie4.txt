#include <iostream>
using namespace std;

int main()
{
    int o1, o2, o3, o4, o5;
    int jedynki = 0;

    cin >> o1 >> o2 >> o3 >> o4 >> o5;

    if (o1 == 1) jedynki++;
    if (o2 == 1) jedynki++;
    if (o3 == 1) jedynki++;
    if (o4 == 1) jedynki++;
    if (o5 == 1) jedynki++;

    double srednia = (o1 + o2 + o3 + o4 + o5) / 5.0;

    cout << "Srednia: " << srednia << endl;
    cout << "Liczba ocen niedostatecznych: " << jedynki;

    return 0;
}
