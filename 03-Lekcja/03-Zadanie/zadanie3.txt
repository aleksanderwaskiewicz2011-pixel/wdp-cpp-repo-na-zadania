#include <iostream>
using namespace std;

int main()
{
    int a, b;

    cin >> a >> b;

    if (b == 0)
    {
        cout << "Nie mozna dzielic przez 0";
    }
    else if (a % b == 0)
    {
        cout << "Pierwsza liczba jest podzielna przez druga";
    }
    else
    {
        cout << "Pierwsza liczba nie jest podzielna przez druga";
    }

    return 0;
}
